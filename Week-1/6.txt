    employee_id INT,
    employee_name VARCHAR(50),
    department VARCHAR(50),
    sales INT

SELECT * FROM sales;

WITH employee_sales AS (
    SELECT 
        employee_id,
        employee_name,
        department,
        SUM(sales) AS total_sales
    FROM sales
    GROUP BY employee_id, employee_name, department
)

SELECT *
FROM employee_sales;

SELECT
    employee_id,
    employee_name,
    department,
    total_sales,
    ROW_NUMBER() OVER (
        PARTITION BY department
        ORDER BY total_sales DESC
    ) AS rank_in_department
FROM employee_sales;


SELECT *
FROM (
    SELECT
        employee_id,
        employee_name,
        department,
        total_sales,
        ROW_NUMBER() OVER (
            PARTITION BY department
            ORDER BY total_sales DESC
        ) AS rank_in_department
    FROM employee_sales
) ranked_employees
WHERE rank_in_department = 1;
